cmake_minimum_required(VERSION 3.21)
project(SeabunnyNextTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Test DBus)
find_package(KF6WindowSystem REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
qt_add_executable(open-with-tests tst_open_with.cpp ../src/open_with.h ../src/open_with.cpp)
target_include_directories(open-with-tests PRIVATE ../src)
target_link_libraries(open-with-tests PRIVATE Qt6::Core Qt6::Test PkgConfig::GIO)
add_test(NAME open-with COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/run_open_with_checks.py --native-only --native-binary $<TARGET_FILE:open-with-tests>)
qt_add_executable(portal-tests tst_portal.cpp ../src/portal_backend.h ../src/portal_backend.cpp)
target_include_directories(portal-tests PRIVATE ../src)
target_compile_definitions(portal-tests PRIVATE SEABUNNY_PORTAL_CONTROLLER_TEST)
target_link_libraries(portal-tests PRIVATE Qt6::Quick Qt6::Test Qt6::DBus KF6::WindowSystem)
add_test(NAME portal COMMAND portal-tests)
set_tests_properties(portal PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

qt_add_executable(thumbnail-tests tst_next_thumbnails.cpp ../cpp/seabunny/thumbnails.h ../cpp/thumbnails.cpp)
target_include_directories(thumbnail-tests PRIVATE ../cpp)
target_link_libraries(thumbnail-tests PRIVATE Qt6::Quick Qt6::Test)
add_test(NAME thumbnails COMMAND thumbnail-tests)
set_tests_properties(thumbnails PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
